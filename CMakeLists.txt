cmake_minimum_required(VERSION 3.16)
project(BeamLab VERSION 0.2.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_executable(beam-worker native/src/worker.cpp)
target_include_directories(beam-worker PRIVATE native/include)
add_executable(beam-tests native/tests/core.cpp)
target_include_directories(beam-tests PRIVATE native/include)
enable_testing()
add_test(NAME numeric-core COMMAND beam-tests)
